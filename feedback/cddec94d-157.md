```
import yaml  # at module level

@staticmethod
def _write_markdown_file(filepath: Path, data: dict) -> None:
    cleaned = {k: v for k, v in data.items() if v is not None}
    yaml_content = yaml.dump(cleaned, default_flow_style=False, sort_keys=False)
    filepath.write_text('---\n' + yaml_content + '---\n')
```
This passes raw typed values to yaml.dump, letting it handle quoting, multi-line strings, and type preservation correctly. Also adds `sort_keys=False` to preserve insertion order.
